function applyAdmin(change){if(change.action==='pause'){if(onlineSession)onlineSession.adminPaused=change.paused;else paused=change.paused;clearInputs();return;}if(change.action==='step'){if(onlineSession)simulate([neutral(),neutral()]);else if(paused)simulate([neutral(),selected.mode==='training'?lab.dummy(game):neutral()]);return;}game.adminApply(change);selected.characters=[...game.characters];selected.colors=[...game.colors];selected.rules={...game.rules};selected.mapId=game.mapId;lab.history=[];lab.clip=null;if(game.mode==='training')lab.startTraining(game);resetEffects();clearInputs();renderer.snapCamera=true;$('match-meta').textContent=MAPS[game.mapId].name+' · TESTE';if(onlineSession){lobby.config={...lobby.config,mapId:game.mapId,rules:cleanRules(game.rules),adminSession:true};lobby.players.forEach((p,i)=>{p.kind=game.characters[i];p.color=game.colors[i];});}}
function requestAdmin(change){if(inMenu||lab.viewGame)return false;if(selected.mode!=='online'){applyAdmin(change);return true;}if(!onlineSession||seat!==0||!lobby?.config.adminSession||onlineSession.pendingAdmin)return false;const frame=onlineSession.adminPaused?onlineSession.frame:onlineSession.frame+60,token=crypto.randomUUID();onlineSession.pendingAdmin={change,frame,token,acked:false};return sendRequired({type:'adminOffer',id:onlineSession.id,change,frame,token});}
function receiveAdmin(msg){const s=onlineSession;if(!s||msg.id!==s.id||!lobby?.config.adminSession)return false;if(msg.type==='adminAck'&&seat===0&&s.pendingAdmin?.token===msg.token){s.pendingAdmin.acked=true;return true;}if(msg.type!=='adminOffer'||seat!==1)return false;if(!Number.isInteger(msg.frame)||msg.frame<s.frame||msg.frame>s.frame+180||typeof msg.token!=='string'||msg.token.length>60||s.pendingAdmin)return true;const c=msg.change;if(!c||!['apply','reset','refill','pause','step'].includes(c.action))return true;if(c.action==='apply'&&(!MAPS[c.mapId]||!Array.isArray(c.characters)||c.characters.length!==2||c.characters.some(k=>!CLASSES[k])||!c.rules||Object.keys(RULES).some(k=>!Number.isFinite(c.rules[k])||c.rules[k]<0)||c.rules.attackSpeed<=0||c.rules.mapSize<=0||c.rules.dashCount<1))return true;s.pendingAdmin={change:c,frame:msg.frame,token:msg.token,acked:true};sendRequired({type:'adminAck',id:s.id,token:msg.token});return true;}
