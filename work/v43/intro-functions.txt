function closeSpecialIntro(){if(introState?.timer)clearInterval(introState.timer);introState=null;if($('special-intro').open)$('special-intro').close();clearInputs();}
function paintIntro(){if(!introState)return;const left=introState.deadline?Math.max(0,Math.ceil((introState.deadline-performance.now())/1000)):10;$('special-countdown').textContent='COMEÇA EM '+left;$('special-understood').disabled=introState.accepted;$('special-understood').textContent=introState.accepted?'PRONTO · AGUARDANDO ADVERSÁRIO':'ENTENDI · EM GUARDA ↗';}
function finishOnlineIntro(){const s=onlineSession;if(!s||s.started||seat!==0)return;if(!sendRequired({type:'go',id:s.id}))return;s.started=true;closeSpecialIntro();s.stalledAt=0;accumulator=0;}
function startIntroClock(){const s=onlineSession;if(seat!==0||!s||s.started||s.introDeadline||!s.peerIntroShown)return;s.introDeadline=performance.now()+10000;if(introState)introState.deadline=s.introDeadline;sendRequired({type:'introClock',id:s.id,remaining:10000});paintIntro();}
function tickIntro(){const s=onlineSession;if(introState?.online){if(!s){closeSpecialIntro();return;}if(s.started){closeSpecialIntro();return;}if(seat===0&&s.introDeadline&&performance.now()>=s.introDeadline){finishOnlineIntro();return;}if(!introState.deadline&&performance.now()-s.createdAt>15000){returnMenu('O adversário não carregou a partida. Reconectem para continuar.');return;}}else if(introState?.deadline&&performance.now()>=introState.deadline){const done=introState.done;closeSpecialIntro();done();return;}paintIntro();}
function presentSpecials(done){closeSpecialIntro();if(!game.rules.specials){done();return;}const content=$('special-cards');content.replaceChildren();for(let i=0;i<2;i++){const f=game.fighters[i],v=DuelSpecialUI.info[f.kind],card=document.createElement('section');card.className='special-card';const label=document.createElement('small');label.textContent=(selected.mode==='local'?'JOGADOR '+(i+1):i===seat?'VOCÊ':'ADVERSÁRIO')+' · '+CLASSES[f.kind].name;const title=document.createElement('h3');title.textContent=v.name;const p=document.createElement('p');p.textContent=f.kind==='boxer'?'Acerte um parry em '+ruleValue('boxerWindow',game.rules.boxerWindow)+' para liberar '+ruleValue('boxerRush',game.rules.boxerRush)+' de socos acelerados sem recuperação. Segure ataque para alternar os punhos.':v.benefit;const cost=document.createElement('p');cost.className='cost';cost.textContent=f.kind==='boxer'?'Se a janela acabar sem parry, fica '+ruleValue('boxerPenalty',game.rules.boxerPenalty)+' sem dash. O alcance continua curto; parry, clash e atordoamento funcionam normalmente.':v.cost;card.append(label,title,p,cost);content.append(card);}
 $('special-intro-note').textContent='Especial: '+input.label(selected.source,'special')+'. Recarga de '+ruleValue('specialCooldown',game.rules.specialCooldown)+'. '+(game.rules.persistSpecial?'Carga preservada entre rodadas; começa vazia em cada partida.':'Carga começa vazia a cada rodada.')+' Recarrega após o efeito e a penalidade.';
 const online=selected.mode==='online';introState={online,accepted:false,deadline:online?0:performance.now()+10000,done,timer:setInterval(tickIntro,100)};$('special-intro').showModal();clearInputs();paintIntro();$('special-understood').onclick=()=>{if(!introState||introState.accepted)return;introState.accepted=true;if(online){done();paintIntro();}else{closeSpecialIntro();done();}};
 if(online&&seat===1)sendRequired({type:'introShown',id:onlineSession.id});
}
function onlineIntroDone(){const s=onlineSession;if(!s||s.started)return;s.introAccepted=true;if(seat===1)sendRequired({type:'ready',id:s.id});else if(s.peerAccepted)finishOnlineIntro();}
