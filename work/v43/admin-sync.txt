// The host orders commands, even in the common edition. Privilege belongs to the build, not the seat.
function adminSeat(id){return id===seat?!!admin:!!net?.remoteAdmin;}
function validAdminChange(c){if(!c||!['apply','reset','refill','pause','step'].includes(c.action))return false;if(c.action==='pause')return typeof c.paused==='boolean';if(c.action!=='apply')return true;
 if(c.mapData){try{if(c.mapData.id!==c.mapId)return false;DuelCore.cleanMap(c.mapData);}catch{return false;}}
 return (!!MAPS[c.mapId]||!!c.mapData)&&Array.isArray(c.characters)&&c.characters.length===2&&c.characters.every(k=>!!CLASSES[k])&&Array.isArray(c.skins)&&c.skins.length===2&&c.skins.every((v,i)=>v===DuelCore.cleanSkin(c.characters[i],v))&&Array.isArray(c.colors)&&c.colors.length===2&&c.colors.every(k=>!!COLORS[k])&&c.colors[0]!==c.colors[1]&&c.rules&&['airDash','specials','persistSpecial'].every(k=>typeof c.rules[k]==='boolean')&&Object.keys(RULES).every(k=>Number.isFinite(c.rules[k])&&c.rules[k]>=0&&(!['attackSpeed','mapSize'].includes(k)||c.rules[k]>0)&&(!['dashCount','winningScore'].includes(k)||Number.isSafeInteger(c.rules[k])&&c.rules[k]>=1))&&Array.isArray(c.positions)&&c.positions.length===2&&c.positions.every(p=>p&&Object.keys(p).every(k=>['x','y'].includes(k)&&Number.isFinite(p[k])));
}
function requestAdmin(change){if(!admin||inMenu||lab.viewGame)return false;if(change.mapId&&MAPS[change.mapId]?.custom)change.mapData=MAPS[change.mapId];if(!validAdminChange(change))return false;if(selected.mode!=='online'){applyAdmin(change);return true;}const s=onlineSession;if(!s?.started)return false;const token=crypto.randomUUID();if(seat===0)return enqueueAdmin(change,0,token);return sendRequired({type:'adminRequest',id:s.id,change,token});}
function enqueueAdmin(change,actor,token){const s=onlineSession;if(!s||seat!==0||!adminSeat(actor)||!validAdminChange(change))return false;if(s.adminTokens.has(token))return true;s.adminTokens.add(token);s.adminQueue.push({change,actor,token});offerNextAdmin();return true;}
function offerNextAdmin(){const s=onlineSession;if(!s||seat!==0||s.pendingAdmin||!s.adminQueue.length)return;const p=s.adminQueue.shift();p.frame=s.adminPaused?s.frame:s.frame+60;p.sequence=++s.adminSequence;p.acked=false;p.applied=false;p.remoteApplied=false;p.sentAt=performance.now();s.pendingAdmin=p;sendRequired({type:'adminOffer',id:s.id,change:p.change,frame:p.frame,token:p.token,sequence:p.sequence,actor:p.actor});}
function finishAdminOffer(){const s=onlineSession,p=s?.pendingAdmin;if(seat===0&&p?.applied&&p.remoteApplied){s.pendingAdmin=null;offerNextAdmin();}}
function receiveAdmin(msg){const s=onlineSession;if(!s||msg.id!==s.id||!msg.type?.startsWith('admin'))return false;
 if(msg.type==='adminRequest'&&seat===0&&net.remoteAdmin&&s.started){if(typeof msg.token==='string'&&msg.token.length<=60)enqueueAdmin(msg.change,1,msg.token);return true;}
 if(msg.type==='adminAck'&&seat===0&&s.pendingAdmin?.token===msg.token){s.pendingAdmin.acked=true;return true;}
 if(msg.type==='adminApplied'&&seat===0&&s.pendingAdmin?.token===msg.token){s.pendingAdmin.remoteApplied=true;finishAdminOffer();return true;}
 if(msg.type!=='adminOffer'||seat!==1)return true;
 if(![0,1].includes(msg.actor)||!adminSeat(msg.actor)||!Number.isInteger(msg.frame)||msg.frame<s.frame||msg.frame>s.frame+180||typeof msg.token!=='string'||msg.token.length>60||msg.sequence!==s.adminSequence+1||s.pendingAdmin||!validAdminChange(msg.change)){returnMenu('Um comando administrativo não pôde ser sincronizado. Reconectem a sala.');return true;}
 s.adminSequence=msg.sequence;s.pendingAdmin={change:msg.change,frame:msg.frame,token:msg.token,sequence:msg.sequence,actor:msg.actor,acked:true,applied:false,sentAt:performance.now()};sendRequired({type:'adminAck',id:s.id,token:msg.token});return true;
}
function adminBarrier(){const s=onlineSession,p=s?.pendingAdmin;if(!p)return false;if(performance.now()-p.sentAt>12000){returnMenu('O comando administrativo perdeu contato com o adversário. Reconectem a sala.');return true;}if(p.applied||s.frame<p.frame)return false;if(!p.acked){accumulator=0;return true;}
 p.applied=true;applyAdmin(p.change);s.lastAdmin={actor:p.actor,sequence:p.sequence,action:p.change.action};effects.eventText='ADMIN · JOGADOR '+(p.actor+1)+' · '+({apply:'REGRAS APLICADAS',pause:p.change.paused?'PAUSA':'RETOMADA',step:'+1 QUADRO',refill:'HABILIDADES RECARREGADAS',reset:'RODADA REINICIADA'}[p.change.action]);effects.eventLife=2.2;s.stalledAt=0;accumulator=0;
 if(seat===1){s.pendingAdmin=null;sendRequired({type:'adminApplied',id:s.id,token:p.token});}else finishAdminOffer();return !!s.adminPaused;
}
