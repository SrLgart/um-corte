function cleanMap(data){if(!data||typeof data!=='object'||data.format!=='um-corte-map'||data.version!==1)throw Error('Arquivo de mapa incompatível.');const number=(v,min,max)=>{if(!Number.isFinite(v)||v<min||v>max)throw Error('Medida do mapa fora dos limites.');return v;},color=v=>/^#[0-9a-f]{6}$/i.test(v)?v:'#607266';
 const width=number(data.width,640,3840),height=number(data.height,360,2160),id=String(data.id||'');if(!/^custom_[a-zA-Z0-9_-]{1,60}$/.test(id))throw Error('Identificador de mapa inválido.');if(!Array.isArray(data.platforms)||data.platforms.length>80||!Array.isArray(data.walls)||data.walls.length>60||!Array.isArray(data.spawns)||data.spawns.length!==2)throw Error('Mapa precisa de dois inícios e até 80 plataformas / 60 paredes.');
 const platforms=data.platforms.map(p=>{const q={x:number(p.x,0,width),y:number(p.y,0,height),w:number(p.w,20,width),solid:!!p.solid};if(q.x+q.w>width+1)throw Error('Plataforma fora da arena.');if(p.motion){q.motion={x:number(p.motion.x,-width,width),y:number(p.motion.y,-height,height),period:number(p.motion.period,.4,120),pause:number(p.motion.pause??0,0,30),route:true};if(q.x+q.motion.x<0||q.x+q.motion.x+q.w>width||q.y+q.motion.y<0||q.y+q.motion.y>height)throw Error('Trajeto fora da arena.');}if(p.fragile)q.fragile={delay:number(p.fragile.delay,.1,30),respawn:number(p.fragile.respawn,0,120)};return q;});
 const walls=data.walls.map(w=>{const q={x:number(w.x,0,width),y:number(w.y,0,height),w:number(w.w,12,width),h:number(w.h,12,height)};if(q.x+q.w>width||q.y+q.h>height)throw Error('Parede fora da arena.');return q;});const spawns=data.spawns.map(p=>({x:number(p.x,22,width-22),y:number(p.y,110,height)}));
 const background=data.background||{},layers=(background.layers||[]).slice(0,2).map(l=>{if(typeof l.image!=='string'||l.image.length>900000||!/^data:image\/(png|jpeg|webp);base64,[A-Za-z0-9+/=]+$/.test(l.image))throw Error('Fundo inválido: use PNG, JPEG ou WebP.');return{image:l.image,parallax:number(l.parallax??.1,0,.5),opacity:number(l.opacity??1,0,1),fit:l.fit==='contain'?'contain':'cover'};});
 return{format:'um-corte-map',version:1,id,custom:true,name:String(data.name||'Minha arena').slice(0,48),description:String(data.description||'Arena criada no editor.').slice(0,160),width,height,deathY:height+170,theme:['dojo','bridge','ruins','bamboo','frost','forge','tower','sanctuary'].includes(data.theme)?data.theme:'dojo',platforms,walls,spawns,ground:color(data.ground),background:{layers,tint:color(background.tint||'#172838'),shade:number(background.shade??.2,0,.85)}};
}
